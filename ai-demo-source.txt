def init_db():
    conn = sqlite3.connect(DB_PATH)

    cursor = conn.cursor()

    # --------------------------------------------------------
    # 对话记录
    # --------------------------------------------------------
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS conversations (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT NOT NULL,
            role TEXT NOT NULL,
            content TEXT NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
    """)

    # --------------------------------------------------------
    # RAG 知识库 Chunk
    #
    # embedding 使用 JSON 字符串保存
    #
    # 例如：
    # [0.123, -0.456, ...]
    # --------------------------------------------------------
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS rag_chunks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            document_name TEXT NOT NULL,
            chunk_index INTEGER NOT NULL,
            content TEXT NOT NULL,
            embedding TEXT NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
    """)

    conn.commit()
    conn.close()


init_db()

@app.get("/sessions")
async def get_sessions():

    conn = sqlite3.connect(DB_PATH)

    cursor = conn.cursor()

    cursor.execute("""
        SELECT
            session_id,
            MIN(created_at) AS created_at
        FROM conversations
        GROUP BY session_id
        ORDER BY MIN(id) DESC
    """)

    rows = cursor.fetchall()

    sessions = []

    for session_id, created_at in rows:

        # 获取第一条用户消息作为会话标题
        cursor.execute("""
            SELECT content
            FROM conversations
            WHERE session_id = ?
              AND role = 'user'
            ORDER BY id ASC
            LIMIT 1
        """, (session_id,))

        first_message = cursor.fetchone()

        if first_message:
            title = first_message[0]
        else:
            title = "新会话"

        # 前端侧边栏不需要特别长的标题
        if len(title) > 40:
            title = title[:40] + "..."

        sessions.append({
            "session_id": session_id,
            "title": title,
            "created_at": created_at
        })

    conn.close()

    return {
        "sessions": sessions
    }

# 12. 文本切分
# ============================================================

def split_text(
    text: str,
    chunk_size: int = CHUNK_SIZE,
    chunk_overlap: int = CHUNK_OVERLAP
):
    """
    将长文档切分成多个 Chunk。

    例如：

    chunk_size = 500
    chunk_overlap = 100

    那么：

    Chunk 1：0 ~ 500
    Chunk 2：400 ~ 900
    Chunk 3：800 ~ 1300

    这样相邻 Chunk 会有部分重叠，
    避免一句话刚好被切断。
    """

    text = text.strip()

    if not text:
        return []

    if chunk_overlap >= chunk_size:
        raise ValueError(
            "CHUNK_OVERLAP 必须小于 CHUNK_SIZE"
        )

    chunks = []

    start = 0
    text_length = len(text)

    while start < text_length:

        end = min(
            start + chunk_size,
            text_length
        )

        chunk = text[start:end].strip()

        if chunk:
            chunks.append(chunk)

        if end >= text_length:
            break

        start = end - chunk_overlap

    return chunks

# 13. 调用 Ollama 获取 Embedding
# ============================================================

async def get_embedding(text: str):

    payload = {
        "model": EMBED_MODEL,
        "prompt": text
    }

    async with httpx.AsyncClient(
        timeout=60
    ) as client:

        response = await client.post(
            f"{OLLAMA_URL}/api/embeddings",
            json=payload
        )

        response.raise_for_status()

        data = response.json()

    embedding = data.get("embedding")

    if not embedding:
        raise RuntimeError(
            "Ollama 没有返回 embedding"
        )

    return embedding

# 14. 计算余弦相似度
# ============================================================

def cosine_similarity(
    vector_a,
    vector_b
):
    if not vector_a or not vector_b:
        return 0.0

    if len(vector_a) != len(vector_b):
        raise ValueError(
            f"向量维度不一致："
            f"{len(vector_a)} != {len(vector_b)}"
        )

    dot_product = sum(
        a * b
        for a, b in zip(vector_a, vector_b)
    )

    norm_a = math.sqrt(
        sum(a * a for a in vector_a)
    )

    norm_b = math.sqrt(
        sum(b * b for b in vector_b)
    )

    if norm_a == 0 or norm_b == 0:
        return 0.0

    return dot_product / (
        norm_a * norm_b
    )

@app.post("/documents/upload")
async def upload_document(
    file: UploadFile = File(...)
):

    # --------------------------------------------------------
    # 检查文件名
    # --------------------------------------------------------

    if not file.filename:
        raise HTTPException(
            status_code=400,
            detail="文件名不能为空"
        )

    original_name = file.filename

    suffix = Path(original_name).suffix.lower()

    allowed_suffixes = {
        ".txt",
        ".md",
        ".markdown"
    }

    if suffix not in allowed_suffixes:
        raise HTTPException(
            status_code=400,
            detail="只支持 TXT / MD / MARKDOWN 文件"
        )

    # --------------------------------------------------------
    # 读取文件
    # --------------------------------------------------------

    content_bytes = await file.read()

    # 5 MB 限制
    max_size = 5 * 1024 * 1024

    if len(content_bytes) > max_size:
        raise HTTPException(
            status_code=400,
            detail="文件大小不能超过 5MB"
        )

    try:
        text = content_bytes.decode("utf-8")
    except UnicodeDecodeError:
        raise HTTPException(
            status_code=400,
            detail="文件必须使用 UTF-8 编码"
        )

    text = text.strip()

    if not text:
        raise HTTPException(
            status_code=400,
            detail="文档内容为空"
        )

    # --------------------------------------------------------
    # 文本切分
    # --------------------------------------------------------

    chunks = split_text(text)

    if not chunks:
        raise HTTPException(
            status_code=400,
            detail="文档切分后没有有效内容"
        )

    # --------------------------------------------------------
    # 保存原始文件
    #
    # 使用 UUID 避免文件名冲突
    # --------------------------------------------------------

    stored_filename = (
        f"{uuid.uuid4().hex}{suffix}"
    )

    stored_path = DOCUMENT_DIR / stored_filename

    try:
        stored_path.write_bytes(content_bytes)
    except Exception as e:
        raise HTTPException(
            status_code=500,
            detail=f"保存文档失败：{e}"
        )

    conn = sqlite3.connect(DB_PATH)

    try:

        cursor = conn.cursor()

        # ----------------------------------------------------
        # 如果同名文档已经存在
        #
        # 删除旧 Chunk
        # ----------------------------------------------------

        cursor.execute("""
            DELETE FROM rag_chunks
            WHERE document_name = ?
        """, (original_name,))

        # ----------------------------------------------------
        # 逐 Chunk 获取 Embedding
        # ----------------------------------------------------

        inserted_chunks = []

        for index, chunk in enumerate(chunks):

            embedding = await get_embedding(chunk)

            cursor.execute("""
                INSERT INTO rag_chunks (
                    document_name,
                    chunk_index,
                    content,
                    embedding
                )
                VALUES (?, ?, ?, ?)
            """, (
                original_name,
                index,
                chunk,
                json.dumps(
                    embedding,
                    ensure_ascii=False
                )
            ))

            inserted_chunks.append({
                "chunk_index": index,
                "content": chunk,
                "embedding_dimension": len(
                    embedding
                )
            })

        conn.commit()

    except Exception as e:

        conn.rollback()

        # 如果 Embedding 过程中失败，
        # 删除刚刚保存的原始文件
        try:
            if stored_path.exists():
                stored_path.unlink()
        except Exception:
            pass

        raise HTTPException(
            status_code=500,
            detail=f"文档处理失败：{e}"
        )

    finally:
        conn.close()

    return {
        "message": "文档上传并处理成功",
        "document_name": original_name,
        "chunk_count": len(chunks),
        "chunks": inserted_chunks
    }

# ============================================================
# RAG 上下文检索辅助函数
#
# 作用：
# 当前问题如果明显依赖上一轮对话，
# 就把最近一条用户问题带入 RAG 检索。
#
# 示例：
#
# 第一轮：
# Docker 是什么？
#
# 第二轮：
# 它有哪些命令？
#
# 实际检索：
#
# Docker 是什么？
# 它有哪些命令？
#
# 这样 Embedding 才能获得 Docker 这个核心实体。
# ============================================================

def build_retrieval_question(
    user_message: str,
    history_rows
):

    current = user_message.strip()

    if not current:
        return current

    # --------------------------------------------------------
    # 判断当前问题是否明显属于上下文追问
    # --------------------------------------------------------

    context_keywords = [
        "它",
        "这个",
        "这个东西",
        "这个技术",
        "这个工具",
        "这个方案",
        "这个项目",
        "这个系统",
        "这个模型",
        "这个服务",
        "这些",
        "那些",
        "有什么优缺点",
        "优缺点",
        "有哪些命令",
        "有哪些",
        "还有什么",
        "还有哪些",
        "怎么用",
        "怎么安装",
        "怎么配置",
        "怎么部署",
        "为什么",
        "原理是什么",
    ]

    need_context = any(
        keyword in current
        for keyword in context_keywords
    )

    if not need_context:
        return current

    # --------------------------------------------------------
    # 寻找最近一条用户消息
    # --------------------------------------------------------

    previous_user_message = None

    # --------------------------------------------------------
    # 寻找“当前问题之前”的最近一条用户消息
    #
    # /chat 会先把当前问题保存到 SQLite，
    # 所以 history_rows 的最后一条 user 消息
    # 实际上就是当前问题本身。
    #
    # 这里必须跳过当前问题，再寻找上一条用户消息。
    # --------------------------------------------------------

    current_skipped = False

    for role, content in reversed(history_rows):

        if role != "user":
            continue

        content = content.strip()

        # 第一个 user 消息就是当前问题，跳过
        if not current_skipped:

            if content == current:

                current_skipped = True
                continue

        # 找到当前问题之前的上一条用户消息
        previous_user_message = content
        break

    if not previous_user_message:
        return current

    # --------------------------------------------------------
    # 避免重复
    # --------------------------------------------------------

    if previous_user_message == current:
        return current

    # --------------------------------------------------------
    # 组合检索问题
    # --------------------------------------------------------

    retrieval_question = (
        f"{previous_user_message}\n"
        f"{current}"
    )

    return retrieval_question



async def retrieve_documents(
    question: str,
    top_k: int = RAG_TOP_K,
    threshold: float = RAG_SIMILARITY_THRESHOLD
):

    # --------------------------------------------------------
    # 第一步：
    # 将用户问题转换成 Embedding
    # --------------------------------------------------------

    question_vector = await get_embedding(
        question
    )

    # --------------------------------------------------------
    # 第二步：
    # 从 SQLite 读取所有 Chunk
    # --------------------------------------------------------

    conn = sqlite3.connect(DB_PATH)

    cursor = conn.cursor()

    cursor.execute("""
        SELECT
            id,
            document_name,
            chunk_index,
            content,
            embedding
        FROM rag_chunks
    """)

    rows = cursor.fetchall()

    conn.close()

    if not rows:
        return []

    # --------------------------------------------------------
    # 第三步：
    # 计算每个 Chunk 与问题的相似度
    # --------------------------------------------------------

    results = []

    for row in rows:

        embedding = json.loads(row[4])

        similarity = cosine_similarity(
            question_vector,
            embedding
        )

        # ----------------------------------------------------
        # Similarity Threshold
        # ----------------------------------------------------

        if similarity >= threshold:

            results.append({
                "id": row[0],
                "document_name": row[1],
                "chunk_index": row[2],
                "content": row[3],
                "similarity": similarity
            })

    # --------------------------------------------------------
    # 第四步：
    # 按相似度从高到低排序
    # --------------------------------------------------------

    results.sort(
        key=lambda x: x["similarity"],
        reverse=True
    )

    # --------------------------------------------------------
    # 第五步：
    # Top-K
    # --------------------------------------------------------

    return results[:top_k]


# ============================================================
# 19. RAG 调试接口
#
# 前端：
# GET /rag/debug?question=xxx
#
# 前端需要：
# embedding_dimension
# max_similarity
# threshold
# gate_pass
# results
# ============================================================

@app.get("/rag/debug")
async def rag_debug(
    question: str
):
    """
    RAG 调试接口。

    检索逻辑与 /chat 完全一致：

    1. 计算所有 Chunk 的相似度
    2. Threshold 过滤
    3. 按相似度从高到低排序
    4. Top-K 截取
    """

    question = question.strip()

    if not question:
        raise HTTPException(
            status_code=400,
            detail="问题不能为空"
        )

    # 获取问题 Embedding
    question_vector = await get_embedding(
        question
    )

    embedding_dimension = len(
        question_vector
    )

    # 读取所有 Chunk
    conn = sqlite3.connect(DB_PATH)

    cursor = conn.cursor()

    cursor.execute("""
        SELECT
            id,
            document_name,
            chunk_index,
            content,
            embedding
        FROM rag_chunks
    """)

    rows = cursor.fetchall()

    conn.close()

    candidates = []

    # 计算所有 Chunk 的相似度
    for row in rows:

        embedding = json.loads(row[4])

        similarity = cosine_similarity(
            question_vector,
            embedding
        )

        candidates.append({
            "id": row[0],
            "document_name": row[1],
            "chunk_index": row[2],
            "content": row[3],
            "similarity": similarity
        })

    # =========================================================
    # 关键修复
    #
    # 与 /chat 的 retrieve_documents() 保持完全一致：
    #
    # 相似度计算
    #       ↓
    # Threshold 过滤
    #       ↓
    # 相似度排序
    #       ↓
    # Top-K
    # =========================================================

    passed_results = [
        item
        for item in candidates
        if item["similarity"] >= RAG_SIMILARITY_THRESHOLD
    ]

    passed_results.sort(
        key=lambda item: item["similarity"],
        reverse=True
    )

    results = passed_results[:RAG_TOP_K]

    # 所有候选 Chunk 中的最高相似度
    max_similarity = max(
        (
            item["similarity"]
            for item in candidates
        ),
        default=0.0
    )

    gate_pass = (
        max_similarity >= RAG_SIMILARITY_THRESHOLD
    )

    return {
        "question": question,
        "embedding_dimension": embedding_dimension,
        "max_similarity": max_similarity,
        "threshold": RAG_SIMILARITY_THRESHOLD,
        "gate_pass": gate_pass,
        "top_k": RAG_TOP_K,
        "candidate_count": len(candidates),
        "threshold_pass_count": len(passed_results),
        "result_count": len(results),
        "results": results
    }



async def rewrite_query(
    user_message: str,
    history_rows
):
    """
    使用 Qwen 将多轮对话中的省略式问题，
    改写成语义完整、适合 RAG 检索的 Query。

    例如：

    用户：
    Docker 是什么？

    下一轮：
    它有什么优缺点？

    Rewrite 后：

    Docker 有什么优缺点？

    注意：
    这里只负责“改写检索问题”，
    不负责最终回答用户。
    """

    current = user_message.strip()

    if not current:
        return current

    # -----------------------------------------------------
    # 找到当前问题之前的最近几轮对话
    #
    # /chat 会先保存当前 user message，
    # 所以 history_rows 最后一条 user 消息
    # 通常就是当前问题。
    #
    # 这里排除当前问题本身。
    # -----------------------------------------------------

    history_for_rewrite = []

    current_skipped = False

    for role, content in reversed(history_rows):

        content = content.strip()

        if not content:
            continue

        if role == "user":

            # 跳过当前问题
            if not current_skipped and content == current:
                current_skipped = True
                continue

            history_for_rewrite.append(
                f"用户：{content}"
            )

        elif role == "assistant":

            history_for_rewrite.append(
                f"助手：{content}"
            )

        # 只取最近几条历史消息，
        # 防止 Query Rewrite 上下文无限增长。
        if len(history_for_rewrite) >= 6:
            break

    history_for_rewrite.reverse()

    history_text = "\n".join(
        history_for_rewrite
    )

    # 没有历史对话时，不需要 Rewrite
    if not history_text:
        return current

    system_prompt = """
你是一个 RAG 检索 Query Rewrite 助手。

你的唯一任务是：
根据最近的对话历史，把用户当前问题改写成一个
语义完整、明确、适合知识库向量检索的问题。

规则：

1. 如果当前问题已经完整明确，直接原样返回。

2. 如果当前问题包含“它、这个、那个、该技术、
   该工具、这个方案”等指代词，
   根据对话历史补充明确的实体。

3. 不要回答用户的问题。

4. 不要解释改写过程。

5. 不要添加知识库中不存在的信息。

6. 只输出最终的检索 Query。

7. 尽量保持用户原问题的意思不变。

例如：

历史：
用户：Docker 是什么？

当前问题：
它有哪些命令？

只输出：
Docker 有哪些命令？
"""

    messages = [
        {
            "role": "system",
            "content": system_prompt.strip()
        },
        {
            "role": "user",
            "content": (
                "最近对话：\n"
                f"{history_text}\n\n"
                "当前问题：\n"
                f"{current}\n\n"
                "请输出改写后的检索 Query。"
            )
        }
    ]

    headers = {
        "Authorization": f"Bearer {DASHSCOPE_API_KEY}",
        "Content-Type": "application/json"
    }

    payload = {
        "model": MODEL_NAME,
        "messages": messages,
        "stream": False,
        "temperature": 0
    }

    try:

        async with httpx.AsyncClient(
            timeout=60.0
        ) as client:

            response = await client.post(
                f"{DASHSCOPE_BASE_URL}/chat/completions",
                headers=headers,
                json=payload
            )

            response.raise_for_status()

            data = response.json()

            rewritten_query = (
                data["choices"][0]["message"]["content"]
                .strip()
            )

            if not rewritten_query:
                return current

            print(
                "========== Query Rewrite =========="
            )
            print(
                f"原始问题：{current}"
            )
            print(
                f"改写结果：{rewritten_query}"
            )
            print(
                "===================================="
            )

            return rewritten_query

    except Exception as e:

        print(
            "========== Query Rewrite 失败 =========="
        )
        print(
            f"错误：{e}"
        )
        print(
            "将使用原始问题继续。"
        )
        print(
            "========================================"
        )

        # Rewrite 失败不能影响原来的 RAG
        return current


def build_rag_prompt(
    question: str,
    retrieved_chunks
):

    context_parts = []

    for index, item in enumerate(
        retrieved_chunks,
        start=1
    ):

        context_parts.append(
            f"[知识片段 {index}]\n"
            f"文档：{item['document_name']}\n"
            f"内容：{item['content']}"
        )

    context = "\n\n".join(context_parts)

    prompt = f"""
你是一个知识库问答助手。

请根据下面提供的【知识库内容】回答用户问题。

====================
回答规则
====================

1. 只能根据【知识库内容】回答。

2. 不要使用知识库之外的预训练知识、常识或网络信息补充答案。

3. 如果多个知识片段都与问题相关，
   可以综合这些知识片段进行回答。

4. 可以对知识库内容进行归纳、总结和简单解释。

5. 回答必须有知识库内容作为依据。

6. 不要因为知识库中的原文没有直接出现用户问题的完整问法，
   就拒绝回答。

7. 如果知识库能够支持回答，
   直接给出答案，不要讨论“是否能够回答”。

8. 回答简洁、直接、准确。

9. 不要解释内部推理过程。

====================
知识库内容
====================

{context}

====================
用户问题
====================

{question}

====================
请直接回答
====================
"""

    return prompt


# ============================================================
# 21. 调用百炼 Qwen 的核心函数
#
# 这里采用：
#
# FastAPI
#     ↓
# async generator
#     ↓
# httpx.stream()
#     ↓
# DashScope SSE
#     ↓
# yield
#     ↓
# 浏览器
#
# 不再使用之前复杂的 Queue + 两层 Generator。
# ============================================================

async def qwen_stream(
    session_id: str,
    messages: list
):

    assistant_content = ""

    current_task = asyncio.current_task()

    # --------------------------------------------------------
    # 登记当前生成任务
    # --------------------------------------------------------

    GENERATION_TASKS[session_id] = current_task

    print(
        f"========== Qwen 生成任务开始：{session_id} =========="
    )

    request_data = {
        "model": MODEL_NAME,
        "messages": messages,
        "stream": True
    }

    try:

        async with httpx.AsyncClient(
            timeout=None
        ) as client:

            print(
                "========== 准备调用百炼 HTTP API =========="
            )

            async with client.stream(
                "POST",
                f"{DASHSCOPE_BASE_URL}/chat/completions",
                headers={
                    "Authorization":
                        f"Bearer {DASHSCOPE_API_KEY}",
                    "Content-Type":
                        "application/json"
                },
                json=request_data
            ) as response:

                print(
                    f"========== 百炼 HTTP 状态码："
                    f"{response.status_code} =========="
                )

                # ------------------------------------------------
                # 如果 API 返回错误
                # ------------------------------------------------

                if response.status_code >= 400:

                    error_body = await response.aread()

                    print(
                        "========== 百炼错误响应 =========="
                    )
                    print(
                        error_body.decode(
                            "utf-8",
                            errors="ignore"
                        )
                    )

                    response.raise_for_status()

                # ------------------------------------------------
                # 逐行读取 SSE
                # ------------------------------------------------

                async for line in response.aiter_lines():

                    if not line:
                        continue

                    # 调试输出
                    print(
                        f"百炼原始 SSE：{line}"
                    )

                    if not line.startswith("data:"):
                        continue

                    data_text = (
                        line[len("data:"):].strip()
                    )

                    # 百炼流结束标记
                    if data_text == "[DONE]":
                        break

                    try:

                        data = json.loads(
                            data_text
                        )

                    except json.JSONDecodeError as e:

                        print(
                            f"百炼 SSE JSON 解析失败：{e}"
                        )

                        continue

                    choices = data.get(
                        "choices",
                        []
                    )

                    if not choices:
                        continue

                    delta = choices[0].get(
                        "delta",
                        {}
                    )

                    content = delta.get(
                        "content",
                        ""
                    )

                    # ------------------------------------------------
                    # 正常字符串形式
                    # ------------------------------------------------

                    if isinstance(
                        content,
                        str
                    ):

                        if content:

                            assistant_content += content

                            yield content

                    # ------------------------------------------------
                    # 某些兼容接口可能返回数组形式
                    # ------------------------------------------------

                    elif isinstance(
                        content,
                        list
                    ):

                        for item in content:

                            if not isinstance(
                                item,
                                dict
                            ):
                                continue

                            text = item.get(
                                "text",
                                ""
                            )

                            if text:

                                assistant_content += text

                                yield text

    except asyncio.CancelledError:

        print(
            f"========== Qwen 生成任务被取消："
            f"{session_id} =========="
        )

        # --------------------------------------------------------
        # 用户点击停止时，如果已经生成了一部分内容，
        # 保存已经生成的部分。
        # --------------------------------------------------------

        if assistant_content.strip():

            save_message(
                session_id,
                "assistant",
                assistant_content
            )

        raise

    except Exception as e:

        print(
            f"========== 百炼请求失败：{e} =========="
        )

        error_text = (
            "\n\n[AI 服务请求失败，请检查百炼配置。]"
        )

        yield error_text

    else:

        # --------------------------------------------------------
        # 正常完成
        # --------------------------------------------------------

        if assistant_content.strip():

            save_message(
                session_id,
                "assistant",
                assistant_content
            )

        print(
            f"========== Qwen 生成完成："
            f"{session_id} =========="
        )

    finally:

        # --------------------------------------------------------
        # 清理任务
        # --------------------------------------------------------

        if (
            GENERATION_TASKS.get(session_id)
            is current_task
        ):

            GENERATION_TASKS.pop(
                session_id,
                None
            )

        print(
            f"========== Qwen 生成任务清理："
            f"{session_id} =========="
        )

@app.post("/chat")
async def chat(
    request: ChatRequest
):

    session_id = request.session_id.strip()
    user_message = request.message.strip()

    if not session_id:
        raise HTTPException(
            status_code=400,
            detail="session_id 不能为空"
        )

    if not user_message:
        raise HTTPException(
            status_code=400,
            detail="message 不能为空"
        )

    # --------------------------------------------------------
    # 检查是否已经有正在生成的任务
    # --------------------------------------------------------

    existing_task = GENERATION_TASKS.get(
        session_id
    )

    if (
        existing_task
        and not existing_task.done()
    ):

        raise HTTPException(
            status_code=409,
            detail="当前会话已有正在生成的任务"
        )

    # --------------------------------------------------------
    # 保存用户消息
    # --------------------------------------------------------

    save_message(
        session_id,
        "user",
        user_message
    )

    # --------------------------------------------------------
    # 获取历史消息
    #
    # RAG 检索现在需要知道最近的对话上下文。
    # --------------------------------------------------------

    conn = sqlite3.connect(DB_PATH)

    cursor = conn.cursor()

    cursor.execute("""
        SELECT role, content
        FROM conversations
        WHERE session_id = ?
        ORDER BY id ASC
    """, (session_id,))

    history_rows = cursor.fetchall()

    conn.close()

    # --------------------------------------------------------
    # Query Rewrite
    #
    # 使用 Qwen 将多轮对话中的省略式问题，
    # 改写成语义完整的 RAG 检索 Query。
    #
    # 例如：
    #
    # 上一轮：Docker 是什么？
    # 当前：它有什么优缺点？
    #
    # Rewrite 后：
    #
    # Docker 有什么优缺点？
    # --------------------------------------------------------

    rewritten_query = await rewrite_query(
        user_message,
        history_rows
    )

    print(
        "========== RAG 检索问题 =========="
    )

    print(
        f"原始问题：{user_message}"
    )

    print(
        f"Rewrite 后检索问题：{rewritten_query}"
    )

    print(
        "=================================="
    )

    # --------------------------------------------------------
    # RAG 检索
    # --------------------------------------------------------

    try:

        retrieved_chunks = (
            await retrieve_documents(
                rewritten_query
            )
        )

    except Exception as e:

        print(
            f"RAG 检索失败：{e}"
        )

        # 这里不能假装是“没有知识”
        # 因为可能实际上是 Embedding 服务故障。
        error_text = (
            "RAG 检索服务异常，请检查 Embedding / Ollama 服务。"
        )

        async def rag_error_generator():

            yield error_text

        return StreamingResponse(
            rag_error_generator(),
            media_type="text/plain; charset=utf-8"
        )

    # --------------------------------------------------------
    # 打印 RAG 调试信息
    # --------------------------------------------------------

    print(
        "============================================"
    )

    print(
        f"/chat 实际 retrieved_chunks 数量："
        f"{len(retrieved_chunks)}"
    )

    if retrieved_chunks:

        print(
            f"/chat 实际最高相似度："
            f"{retrieved_chunks[0]['similarity']:.4f}"
        )

    else:

        print(
            "/chat 实际没有检索到任何 Chunk"
        )

    print(
        "========== RAG 闸门 =========="
    )

    print(
        f"最高相似度："
        f"{retrieved_chunks[0]['similarity']:.4f}"
        if retrieved_chunks
        else "最高相似度：0.0000"
    )

    print(
        f"最低通过阈值："
        f"{RAG_SIMILARITY_THRESHOLD:.4f}"
    )

    if retrieved_chunks:

        print(
            "✓ 最高相似度达到阈值"
        )

        print(
            "✓ 允许进入百炼大语言模型"
        )

    else:

        print(
            "✕ 最高相似度没有达到阈值"
        )

        print(
            "✕ 拒绝进入百炼大语言模型"
        )

    print(
        "============================================"
    )

    # --------------------------------------------------------
    # RAG Gate：
    #
    # 没有通过 Threshold
    # → 不调用 Qwen
    # → 直接返回拒答
    # --------------------------------------------------------

    if not retrieved_chunks:

        refusal_text = (
            "知识库中没有找到相关信息。"
        )

        async def refusal_generator():

            save_message(
                session_id,
                "assistant",
                refusal_text
            )

            yield refusal_text

        return StreamingResponse(
            refusal_generator(),
            media_type="text/plain; charset=utf-8"
        )

    # --------------------------------------------------------
    # 构建 RAG System Prompt
    # --------------------------------------------------------

    rag_prompt = build_rag_prompt(
        user_message,
        retrieved_chunks
    )

    # --------------------------------------------------------
    # 构建发送给 Qwen 的 messages
    #
    # history_rows 已经在 RAG 检索之前读取，
    # 这里直接复用，避免重复查询 SQLite。
    # --------------------------------------------------------

    messages = [
        {
            "role": "system",
            "content": rag_prompt
        }
    ]

    for role, content in history_rows:

        messages.append({
            "role": role,
            "content": content
        })

    # --------------------------------------------------------
    # 检查 API Key
    # --------------------------------------------------------

    if not DASHSCOPE_API_KEY:

        raise HTTPException(
            status_code=500,
            detail="DASHSCOPE_API_KEY 未配置"
        )

    # --------------------------------------------------------
    # 打印最终发送内容
    # --------------------------------------------------------

    print(
        "========== 发送给百炼的 messages =========="
    )

    for index, message in enumerate(
        messages,
        start=1
    ):

        print(
            f"【消息 {index}】"
        )

        print(
            f"role: {message['role']}"
        )

        print(
            message["content"]
        )

        print(
            "--------------------------------------------"
        )

    print(
        "========== 已完成百炼请求准备 =========="
    )

    # --------------------------------------------------------
    # 直接返回单层异步 Generator
    #
    # 这是本次最重要的结构调整。
    # --------------------------------------------------------

    return StreamingResponse(
        qwen_stream(
            session_id,
            messages
        ),
        media_type="text/plain; charset=utf-8"
    )

@app.post("/stop/{session_id}")
async def stop_generation(
    session_id: str
):

    task = GENERATION_TASKS.get(
        session_id
    )

    # 当前没有任务
    if task is None:

        return {
            "message":
                "当前没有正在生成的任务",
            "session_id":
                session_id,
            "stopped":
                False
        }

    # 任务已经结束
    if task.done():

        GENERATION_TASKS.pop(
            session_id,
            None
        )

        return {
            "message":
                "生成任务已经结束",
            "session_id":
                session_id,
            "stopped":
                False
        }

    # --------------------------------------------------------
    # 取消任务
    # --------------------------------------------------------

    print(
        f"========== 收到停止请求："
        f"{session_id} =========="
    )

    task.cancel()

    try:

        await task

    except asyncio.CancelledError:

        print(
            f"========== 生成任务取消完成："
            f"{session_id} =========="
        )

    return {
        "message":
            "生成任务已取消",
        "session_id":
            session_id,
        "stopped":
            True
    }

import json
import urllib.request


# Ollama Chat API 地址
url = "http://127.0.0.1:11434/api/chat"


# --------------------------------
# 保存整个聊天历史
# --------------------------------
messages = []


# --------------------------------
# 开始循环，让用户可以连续聊天
# --------------------------------
while True:

    # 获取用户输入
    prompt = input("\n你：")

    # 输入 exit 退出程序
    if prompt.lower() == "exit":
        print("程序结束。")
        break


    # --------------------------------
    # 把用户的问题加入聊天历史
    # --------------------------------
    messages.append({
        "role": "user",
        "content": prompt
    })


    # --------------------------------
    # 准备发送给 Ollama 的数据
    # --------------------------------
    data = {
        "model": "qwen2.5:0.5b",

        # 把完整聊天历史发送给模型
        "messages": messages,

        # false = 一次性返回完整回答
        "stream": False
    }


    # Python 字典 → JSON
    json_data = json.dumps(
        data,
        ensure_ascii=False
    ).encode("utf-8")


    # --------------------------------
    # 创建 HTTP POST 请求
    # --------------------------------
    request = urllib.request.Request(
        url,
        data=json_data,
        headers={
            "Content-Type": "application/json"
        },
        method="POST"
    )


    # --------------------------------
    # 发送请求
    # --------------------------------
    with urllib.request.urlopen(request) as response:

        # 读取 Ollama 返回的数据
        raw_data = response.read()

        # 二进制 → 字符串
        text_data = raw_data.decode("utf-8")

        # JSON → Python 字典
        result = json.loads(text_data)


    # --------------------------------
    # 从返回结果中提取 AI 回答
    # --------------------------------
    answer = result["message"]["content"]


    # 显示 AI 回答
    print("\nAI：")

async function deleteSession(id){

  if(!confirm("确定删除这个历史会话吗？删除后无法恢复。")){
    return;
  }

  try{

    const r=await fetch(

        data.detail||"删除失败"

      e.message||
      "删除会话失败，请检查 API 服务。"
    );

  }
}

==================== README.md ====================
# AI 私有化知识库问答系统

基于 FastAPI + Docker + Ollama + Embedding + RAG + 通义千问构建的 AI 知识库问答 Demo。

项目重点模拟企业内部 AI 私有化部署、知识库问答、系统集成与运维场景。

---

## 1. 项目简介

本项目实现了一个可部署的 AI 知识库问答系统。

用户可以：

- 创建会话
- 进行多轮 AI 对话
- 上传知识库文档
- 对知识库进行语义检索
- 基于检索结果生成回答
- 查看历史会话
- 删除知识库文档
- 实时流式接收 AI 输出
- 手动停止正在生成的回答
- 查看系统健康状态
- 调试 RAG 检索结果

项目重点不是单纯调用大模型 API，而是完整实现：

用户
→ Web 前端
→ FastAPI
→ RAG 检索
→ Ollama Embedding
→ 百炼 Qwen
→ 流式返回

这一套完整链路。

## 4. RAG 工作流程
文档上传
上传 TXT / Markdown
        ↓
文本解析
        ↓
文档切块
        ↓
Embedding
        ↓
保存 SQLite

当前默认参数：

CHUNK_SIZE=500
CHUNK_OVERLAP=100
用户提问
用户问题
   ↓
问题 Embedding
   ↓
Cosine Similarity
   ↓
按照相似度排序
   ↓
取 Top-K
   ↓
相似度闸门
   ↓
通过 → 构建 RAG Prompt
   ↓
调用 Qwen
   ↓
流式返回

如果最高相似度低于：

RAG_SIMILARITY_THRESHOLD=0.63

系统不会调用大模型，而是直接返回：

知识库中没有找到相关信息。

这样可以降低模型脱离知识库自行回答的风险。
